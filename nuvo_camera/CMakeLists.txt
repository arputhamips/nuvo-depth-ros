cmake_minimum_required(VERSION 3.25)
project(nuvo_camera LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(nuvo_depth_client_vendor REQUIRED)
find_package(nuvo_depth_client CONFIG REQUIRED)
set(camera_dependencies rclcpp rclcpp_components rcl_interfaces std_msgs sensor_msgs
  stereo_msgs diagnostic_msgs rosgraph_msgs nuvo_camera_msgs)
foreach(dependency IN LISTS camera_dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(nuvo_camera_component SHARED src/conversion.cpp src/camera.cpp src/policy.cpp)
target_compile_features(nuvo_camera_component PUBLIC cxx_std_20)
target_include_directories(nuvo_camera_component PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(nuvo_camera_component nuvo::client)
ament_target_dependencies(nuvo_camera_component ${camera_dependencies})
target_compile_options(nuvo_camera_component PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_node(nuvo_camera_component PLUGIN "nuvo_camera::NuvoCamera" EXECUTABLE nuvo_camera_node)
install(TARGETS nuvo_camera_component EXPORT export_nuvo_camera
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_nuvo_camera HAS_LIBRARY_TARGET)
ament_export_dependencies(nuvo_depth_client_vendor ${camera_dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(nuvo_conversion_test test/conversion_test.cpp)
  target_link_libraries(nuvo_conversion_test nuvo_camera_component)
  ament_add_gtest(nuvo_camera_test test/camera_test.cpp)
  target_link_libraries(nuvo_camera_test nuvo_camera_component)
  ament_add_gtest(nuvo_policy_test test/policy_test.cpp)
  target_link_libraries(nuvo_policy_test nuvo_camera_component)
endif()
ament_package()
